ip-tracker

ipx() {
  export ipx="$1"
  
  if grep -q '^export ipx=' ~/.bashrc; then
    sed -i "s/^export ipx=.*/export ipx=$1/" ~/.bashrc
    else
      echo "export ipx=$1" >> ~/.bashrc
      fi
      
      echo "[+] ipx set to $ipx"
}
