Python Pattern Recognition

by j4ck13

A mental database of rudimentary patterns to choose from when writing Python.

Core Process

  1. Define the problem.
  2. Choose data types and data structures.
  3. Design the program flow.
  4. Identify normal cases and failure cases.
  5. Choose required imports.
  6. Write a simple first version.
  7. Test it with normal, empty, invalid, and unexpected input.
  8. Add logging and defensive error handling.

from collections import deque

def breadth_first_search(graph, start, target):

    queue = deque([start])

    visited = {start}

    while queue:

        current = queue.popleft()

        if current == target:

            return True

        for neighbor in graph.get(current, []):

            if neighbor not in visited:

                visited.add(neighbor)

                queue.append(neighbor)

    return False

def depth_first_search(graph, current, target, visited=None):

    if visited is None:

        visited = set()

    if current == target:

        return True

    visited.add(current)

    for neighbor in graph.get(current, []):

        if neighbor not in visited:

            if depth_first_search(graph, neighbor, target, visited):

                return True

    return False

Testing Checklist

  • Normal input
  • Empty input
  • Very large input
  • Negative input
  • Wrong data types
  • Missing files
  • Permission failures
  • Duplicate values
  • Unexpected formatting
  • Network or external-service failures

Defensive Programming Questions

  • What if the input is empty?
  • What if the input has the wrong type?
  • What if the file does not exist?
  • What if access is denied?
  • What if a value is outside the allowed range?
  • What if an external command fails?
  • What if the network is unavailable?
  • What information should not appear in an error message?
  • What should be logged?
  • Can the operation be safely repeated?

Small Program Template

import argparse

import logging

from pathlib import Path


def read_file(path: Path) -> str:

    """Read a UTF-8 text file."""

    try:

        return path.read_text(encoding="utf-8")

    except FileNotFoundError:

        logging.error("File was not found: %s", path)

        raise

    except PermissionError:

        logging.error("Permission denied: %s", path)

        raise


def analyze_text(text: str) -> dict:

    """Return basic statistics about text."""

    lines = text.splitlines()

    error_lines = [line for line in lines if "ERROR" in line]

    return {

        "line_count": len(lines),

        "error_count": len(error_lines),

    }


def main():

    parser = argparse.ArgumentParser(description="Analyze a text file.")

    parser.add_argument("filename")

    parser.add_argument("--verbose", action="store_true")

    args = parser.parse_args()

    logging.basicConfig(

        level=logging.DEBUG if args.verbose else logging.INFO,

        format="%(asctime)s [%(levelname)s] %(message)s"

    )

    path = Path(args.filename)

    try:

        text = read_file(path)

        results = analyze_text(text)

        print(f"Lines: {results['line_count']}")

        print(f"Errors: {results['error_count']}")

    except OSError:

        print("The file could not be processed.")


if __name__ == "__main__":

    main()
